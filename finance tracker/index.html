Sinkronisasi Google Sheet
Tempel URL hasil deploy Apps Script (langkah ada di bawah). Kosongkan & simpan untuk mematikan sinkronisasi.
Cara menghubungkan Google Sheet (sekali saja, ±5 menit)
  1. Buka Google Sheets dan buat spreadsheet baru (bebas dinamai apa saja), misal "Buku Kas Data".
  2. Di spreadsheet itu, klik menu Extensions / Ekstensi → Apps Script.
  3. Hapus kode default di editor, lalu tempel kode berikut:
const SHEET_NAME = "Sheet1";
// Header baku yang ditulis ulang otomatis tiap kali menyimpan (self-healing).
const HEADERS = ["id","tanggal","hari","deskripsi","kategori","tipe","jumlah","metode","jam","bulan","tahun","ts"];
// Nama alternatif kolom yang tetap dikenali kalau header di sheet berbeda/manual.
const HEADER_ALIASES = {
  id: ["id"], tanggal: ["tanggal","date"], hari: ["hari","day"],
  deskripsi: ["deskripsi","description","keterangan","catatan"], kategori: ["kategori","category"],
  tipe: ["tipe","type"], jumlah: ["jumlah","amount","nominal","total"], metode: ["metode","method","sumber dana","pembayaran"],
  jam: ["jam","time","waktu"], bulan: ["bulan","month"], tahun: ["tahun","year"],
  ts: ["ts","timestamp"]
};
const HARI_ID = ["Minggu","Senin","Selasa","Rabu","Kamis","Jumat","Sabtu"];
const BULAN_ID = ["Januari","Februari","Maret","April","Mei","Juni","Juli","Agustus",
                  "September","Oktober","November","Desember"];
const KATEGORI_PEMASUKAN = ["Gaji","Bonus","Hadiah","Investasi","Lainnya"];
const KATEGORI_PENGELUARAN = ["Makanan","Transport","Belanja","Pendidikan","Hiburan","Kesehatan","Lainnya"];

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
  }
  return sheet;
}

// Bersihkan teks header: buang spasi tak terlihat (NBSP/zero-width/BOM yang
// suka nyangkut kalau header di-copy-paste dari tempat lain), lalu lowercase.
function cleanHeaderText_(h) {
  return String(h == null ? "" : h)
    .replace(/[\u00A0\u200B\uFEFF]/g, " ")
    .trim()
    .toLowerCase();
}

// Memetakan kolom berdasarkan NAMA header (bebas urutan/besar-kecil huruf),
// bukan posisi tetap -- supaya tetap terbaca walau header ditulis manual.
function buildColumnMap_(headerRow) {
  const map = {};
  headerRow.forEach((h, i) => {
    const key = cleanHeaderText_(h);
    if (!key) return;
    for (const field in HEADER_ALIASES) {
      if (HEADER_ALIASES[field].indexOf(key) !== -1) map[field] = i;
    }
  });
  // Self-healing: kalau TIDAK ADA satupun header yang cocok (mis. baris 1
  // ternyata bukan baris header, atau labelnya beda total dari daftar alias
  // di atas), jangan sampai semua kolom hilang -- jatuh balik ke urutan
  // posisi HEADERS supaya data tetap kebaca selama urutan kolomnya sama
  // dengan yang ditulis otomatis oleh getSheet_().
  if (Object.keys(map).length === 0) {
    HEADERS.forEach((field, i) => { map[field] = i; });
  }
  return map;
}

function parseTanggalJam_(tanggal, jam) {
  // Google Sheets sering otomatis mengubah sel yang diketik "dd/mm/yyyy"
  // menjadi tipe Date asli (bukan teks). Kalau begitu, baca langsung dari
  // objek Date-nya, bukan dari hasil String() yang formatnya tidak cocok
  // dengan regex di bawah (ini penyebab umum Tanggal/Hari/Bulan jadi kosong).
  if (Object.prototype.toString.call(tanggal) === "[object Date]" && !isNaN(tanggal.getTime())) {
    const dt = new Date(tanggal.getFullYear(), tanggal.getMonth(), tanggal.getDate());
    if (Object.prototype.toString.call(jam) === "[object Date]" && !isNaN(jam.getTime())) {
      dt.setHours(jam.getHours(), jam.getMinutes(), 0, 0);
    } else {
      const jm = String(jam || "00:00").trim().match(/^(\d{1,2}):(\d{1,2})/);
      if (jm) dt.setHours(parseInt(jm[1],10), parseInt(jm[2],10), 0, 0);
    }
    return dt;
  }
  const m = String(tanggal || "").trim().match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (!m) return null;
  const d = parseInt(m[1],10), mo = parseInt(m[2],10) - 1, y = parseInt(m[3],10);
  const jm = String(jam || "00:00").trim().match(/^(\d{1,2}):(\d{1,2})/);
  const h = jm ? parseInt(jm[1],10) : 0, mi = jm ? parseInt(jm[2],10) : 0;
  const dt = new Date(y, mo, d, h, mi, 0);
  return isNaN(dt.getTime()) ? null : dt;
}

// Menghitung ulang Hari/Bulan/Tahun/ts dari Tanggal+Jam (satu-satunya
// sumber kebenaran), supaya benar walau ditulis manual dg format berbeda
// (mis. "Thursday", "januari"), dan memperbaiki Kategori yang kosong /
// tertukar dengan nilai Tipe.
function normalizeRow_(obj) {
  const dt = parseTanggalJam_(obj.tanggal, obj.jam);
  // Terima juga Jumlah yang ditulis/diformat sebagai teks mata uang,
  // mis. "Rp 50.000" atau "50,000" -- buang semua karakter selain angka/koma/titik.
  obj.jumlah = Number(String(obj.jumlah == null ? "" : obj.jumlah).replace(/[^\d.-]/g, "")) || 0;
  obj.tipe = String(obj.tipe || "").trim().toLowerCase() === "pemasukan" ? "Pemasukan" : "Pengeluaran";
  if (dt) {
    obj.hari = HARI_ID[dt.getDay()];
    obj.bulan = BULAN_ID[dt.getMonth()];
    obj.tahun = String(dt.getFullYear());
    obj.ts = dt.getTime();
    // Tulis ulang Tanggal & Jam ke format baku aplikasi (dd/mm/yyyy & HH:mm)
    // supaya nilainya konsisten dipakai balik oleh client, bukan objek Date
    // mentah yang kalau di-JSON-kan jadi string ISO yang tidak dikenali client.
    obj.tanggal = Utilities.formatDate(dt, Session.getScriptTimeZone(), "dd/MM/yyyy");
    obj.jam = Utilities.formatDate(dt, Session.getScriptTimeZone(), "HH:mm");
  } else {
    obj.ts = Number(obj.ts) || 0;
    obj.tanggal = String(obj.tanggal || "");
    obj.jam = String(obj.jam || "00:00");
  }
  let kategori = String(obj.kategori || "").trim();
  if (!kategori || kategori.toLowerCase() === obj.tipe.toLowerCase()) kategori = "Lainnya";
  const daftar = obj.tipe === "Pemasukan" ? KATEGORI_PEMASUKAN : KATEGORI_PENGELUARAN;
  const cocok = daftar.find(k => k.toLowerCase() === kategori.toLowerCase());
  obj.kategori = cocok || kategori;
  obj.deskripsi = String(obj.deskripsi == null ? "" : obj.deskripsi).trim();
  obj.metode = String(obj.metode == null ? "" : obj.metode).trim();
  obj.id = obj.id ? String(obj.id) : Utilities.getUuid();
  return obj;
}

function doGet(e) {
  const sheet = getSheet_();
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) {
    return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON);
  }
  const colMap = buildColumnMap_(values[0]);
  const rows = values.slice(1).filter(r => r.join("") !== "");
  const data = rows.map(r => {
    const obj = {};
    for (const field in colMap) obj[field] = r[colMap[field]];
    return normalizeRow_(obj);
  });
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = getSheet_();
  const body = JSON.parse(e.postData.contents).map(normalizeRow_);
  sheet.clearContents();
  sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
  if (body.length > 0) {
    const rows = body.map(t => HEADERS.map(h => (t[h] !== undefined ? t[h] : "")));
    sheet.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", count: body.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
  1. Klik ikon Simpan (💾) di Apps Script.
  2. Klik tombol biru Deploy → New deployment. Pilih tipe Web app.
  3. Isi "Execute as" = Me, dan "Who has access" = Anyone. Klik Deploy.
  4. Google akan minta izin akses — setujui (klik akun kamu → Advanced/Lanjutan → Go to project (unsafe) → Allow, ini aman karena scriptnya milikmu sendiri).
  5. Salin Web app URL yang muncul (diakhiri /exec), tempel di kolom URL di atas, lalu klik Simpan Pengaturan.

Setelah aktif, setiap transaksi otomatis tersimpan ke Google Sheet ini. Data lokal di browser tetap dipakai sebagai cadangan kalau koneksi internet terputus.

Sudah pernah deploy sebelumnya? Kode di atas sudah diperbarui (kini otomatis mengoreksi Hari/Bulan/Tahun/Kategori yang salah format).
